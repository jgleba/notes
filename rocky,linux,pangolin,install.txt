SERVICE

Enable as a systemd service (optional but recommended)

sudo nano /etc/systemd/system/pangolin.service

ADD

[Unit]
Description=Pangolin Service
After=network.target

[Service]
Type=simple
User=pangolin
WorkingDirectory=/opt/pangolin
ExecStart=/usr/bin/python3 -m pangolin serve --host 0.0.0.0 --port 8080
Restart=always

[Install]
WantedBy=multi-user.target

ENABLE SERVICE

sudo systemctl daemon-reload
sudo systemctl enable --now pangolin
sudo systemctl status pangolin
