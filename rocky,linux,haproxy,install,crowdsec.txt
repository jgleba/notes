#sudo dnf install crowdsec crowdsec-firewall-bouncer
sudo dnf install crowdsec crowdsec-firewall-bouncer-iptables -y

ENABLE
sudo systemctl enable --now crowdsec
sudo systemctl enable --now crowdsec-firewall-bouncer

INSTALL MORE - LATEST DID NOTHING 
sudo cscli collections install crowdsecurity/haproxy

GET STATS
sudo cscli metrics

sudo cscli alerts list

sudo cscli decisions list

==

TO MANUALLY BLOCK

sudo cscli decisions add --ip IPADDRESS
sudo cscli decisions delete --ip IPADDRESS

==
REGISTER BOUNCER

sudo cscli bouncers add firewall-bouncer

API key for 'firewall-bouncer':
Please keep this key since you will not be able to retrieve it!

ADD API KEY
sudo nano /etc/crowdsec/bouncers/crowdsec-firewall-bouncer.yaml

RESTART
sudo systemctl restart crowdsec-firewall-bouncer
sudo systemctl status crowdsec-firewall-bouncer

sudo nano /etc/crowdsec/notifications/email.yaml

ALSO ADD

ADD EMAIL NOTIFICATION AS OPTION USING BELOW

sudo nano /etc/crowdsec/profiles.yaml

Find your active profile (usually this one):

name: default_ip_remediation

ADD AFTER DECISIONS:

notifications:
  - email_default

Example full profile:

name: default_ip_remediation
filters:
  - Alert.Remediation == true
decisions:
  - type: ban
    duration: 4h
notifications:
  - email_default

CHANGE THIS IN BOTH SECTIONS OF THE FILE

sudo systemctl restart crowdsec-firewall-bouncer


test
